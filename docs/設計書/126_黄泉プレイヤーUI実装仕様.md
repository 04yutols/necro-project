/** 10階刻み（floor%10===0）かつ初回クリアの場合にのみ階層番号を返す。それ以外 null（=配信しない）。 */
export function getYomiMilestoneFloorForFirstClear(

  const floor = getYomiFloorNumber(stageId);

}
